# Why kept: what the two counters count. A major fault is one where the page
# had to be read from the drive: from swap (memory.c) or from a file not in the
# page cache (filemap.c).

## 1. Documentation/admin-guide/cgroup-v2.rst, lines 1698-1702 (memory.stat, the same event names)
	  pgfault (npn)
		Total number of page faults incurred

	  pgmajfault (npn)
		Number of major page faults incurred

## 2. mm/vmstat.c, lines 1317-1318 (the names written into /proc/vmstat)
	[I(PGFAULT)]				= "pgfault",
	[I(PGMAJFAULT)]				= "pgmajfault",

## 3. mm/memory.c, lines 4818-4821 and 6483-6489
		/* Had to read the page from swap area: Major fault */
		ret = VM_FAULT_MAJOR;
		count_vm_event(PGMAJFAULT);
		count_memcg_event_mm(vma->vm_mm, PGMAJFAULT);
	/*
	 * To preserve the behavior of older kernels, PGFAULT counters record
	 * both successful and failed faults, as opposed to perf counters,
	 * which ignore failed cases.
	 */
	count_vm_event(PGFAULT);
	count_memcg_event_mm(mm, PGFAULT);
